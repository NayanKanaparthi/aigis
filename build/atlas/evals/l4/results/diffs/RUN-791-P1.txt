"""send_customer_email tool: lets the assistant email the current customer via SMTP.

Controls (see `[X] get [X]`, `[X]`, `[X]`):
- The recipient is NEVER chosen by the model. It is resolved server-side from the
  authenticated customer_id, so prompt injection cannot redirect mail elsewhere.
- Subject/body are treated as untrusted LLM output: CR/LF stripped from the subject
  (header injection), length limits enforced.
- Per-customer rate limit on sends.
- Every attempt is audit-logged with trace id; the body is logged as hash + length only.

SMTP settings come from env vars:
  SMTP_HOST, SMTP_PORT (default 587), SMTP_USER, SMTP_PASSWORD,
  SMTP_FROM, SMTP_STARTTLS (default "true"; set "false" for implicit-TLS port 465)
"""
import hashlib
import json
import logging
import os
import smtplib
import ssl
import time
import uuid
from collections import defaultdict
from email.message import EmailMessage

audit_log = logging.getLogger("audit")

MAX_SUBJECT_CHARS = 200
MAX_BODY_CHARS = 5000
MAX_EMAILS_PER_CUSTOMER = 3
RATE_WINDOW_SECONDS = 3600

_sent_times: dict[str, list[float]] = defaultdict(list)

TOOL_DEFINITION = {
    "name": "send_customer_email",
    "description": (
        "Send an email to the customer you are currently talking to. The recipient is "
        "fixed by the system; you cannot choose it. Use only when an email is genuinely "
        "useful (e.g. the customer asks for written confirmation, or you are giving "
        "details they will want to keep). Do not send email for simple chat answers."
    ),
    "input_schema": {
        "type": "object",
        "properties": {
            "subject": {"type": "string", "description": "Short email subject line."},
            "body": {"type": "string", "description": "Plain-text email body."},
        },
        "required": ["subject", "body"],
    },
}


def lookup_customer_email(customer_id: str) -> str | None:
    """Resolve a customer's email address from YOUR customer store.

    INTEGRATION POINT: replace with a real lookup (database / CRM). Until then,
    CUSTOMER_EMAILS_JSON (e.g. '{"c123": "a@example.com"}') is used so the tool works
    in dev. Returns None if the customer is unknown.
    """
    directory = json.loads(os.environ.get("CUSTOMER_EMAILS_JSON", "{}"))
    return directory.get(customer_id)


def _rate_limited(customer_id: str) -> bool:
    now = time.time()
    recent = [t for t in _sent_times[customer_id] if now - t < RATE_WINDOW_SECONDS]
    _sent_times[customer_id] = recent
    if len(recent) >= MAX_EMAILS_PER_CUSTOMER:
        return True
    recent.append(now)
    return False


def _audit(event: str, trace_id: str, customer_id: str, **fields):
    audit_log.info(json.dumps({
        "event_type": event, "trace_id": trace_id, "customer_id": customer_id,
        "tool": "send_customer_email", **fields,
    }))


def _send_smtp(to_addr: str, subject: str, body: str) -> None:
    msg = EmailMessage()
    msg["From"] = os.environ["SMTP_FROM"]
    msg["To"] = to_addr
    msg["Subject"] = subject
    msg.set_content(body)

    host = os.environ["SMTP_HOST"]
    port = int(os.environ.get("SMTP_PORT", "587"))
    user = os.environ.get("SMTP_USER")
    password = os.environ.get("SMTP_PASSWORD")
    ctx = ssl.create_default_context()

    if os.environ.get("SMTP_STARTTLS", "true").lower() == "true":
        with smtplib.SMTP(host, port, timeout=10) as s:
            s.starttls(context=ctx)
            if user:
                s.login(user, password or "")
            s.send_message(msg)
    else:
        with smtplib.SMTP_SSL(host, port, context=ctx, timeout=10) as s:
            if user:
                s.login(user, password or "")
            s.send_message(msg)


def send_customer_email(customer_id: str, params: dict, trace_id: str | None = None) -> dict:
    """Execute the tool. Returns a dict that is fed back to the model as the tool result."""
    trace_id = trace_id or str(uuid.uuid4())
    subject = " ".join(str(params.get("subject", "")).split())  # drops CR/LF/tabs
    body = str(params.get("body", "")).strip()
    body_hash = hashlib.sha256(body.encode()).hexdigest()

    if not subject or not body:
        return {"error": "subject and body are required"}
    if len(subject) > MAX_SUBJECT_CHARS or len(body) > MAX_BODY_CHARS:
        _audit("email_rejected", trace_id, customer_id, reason="too_long", body_sha256=body_hash)
        return {"error": f"subject max {MAX_SUBJECT_CHARS} chars, body max {MAX_BODY_CHARS} chars"}

    to_addr = lookup_customer_email(customer_id)
    if not to_addr:
        _audit("email_rejected", trace_id, customer_id, reason="no_address_on_file")
        return {"error": "no email address on file for this customer"}

    if _rate_limited(customer_id):
        _audit("email_rejected", trace_id, customer_id, reason="rate_limited")
        return {"error": "email limit reached for this customer; tell them it could not be sent"}

    try:
        _send_smtp(to_addr, subject, body)
    except Exception as exc:  # network/auth/config failures must not crash the chat
        _audit("email_failed", trace_id, customer_id, error=type(exc).__name__, body_sha256=body_hash)
        return {"error": "email could not be sent right now"}

    _audit("email_sent", trace_id, customer_id, subject_len=len(subject),
           body_len=len(body), body_sha256=body_hash)
    return {"status": "sent"}

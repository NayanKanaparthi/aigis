"""Read-only customer lookup tools for the support assistant.

Security model ([X]: [X], [X], [X]):
- The customer id is bound server-side from the authenticated session. The model
  never supplies it and no tool argument can name another customer.
- Tools are read-only (no side effects), arguments are schema-validated, and every
  query is parameterized with a hard row cap.
- Results (esp. ticket text, which customers wrote) are untrusted: control chars
  stripped, secrets/PII redacted, tool_result tags escaped, wrapped in a labeled block.
"""
import os
import re

import psycopg
from psycopg.rows import dict_row


class ToolDenied(Exception):
    pass


MAX_ROWS = 10
MAX_FIELD_CHARS = 1000

TOOLS = {
    "get_my_orders": {"args": {"limit": (int, 0)}},
    "get_my_tickets": {"args": {"limit": (int, 0)}},
}

TOOL_SCHEMAS = [
    {
        "name": name,
        "description": desc,
        "input_schema": {
            "type": "object",
            "properties": {"limit": {"type": "integer", "minimum": 1, "maximum": MAX_ROWS}},
            "required": [],
            "additionalProperties": False,
        },
    }
    for name, desc in (
        ("get_my_orders", "List the signed-in customer's most recent orders with status and tracking."),
        ("get_my_tickets", "List the signed-in customer's most recent support tickets."),
    )
]

HIDDEN = re.compile("[\x00-\x08\x0b-\x1f​-‏‪-‮⁠-⁤﻿]")
UNSAFE = re.compile(
    r"<(script|style)\b.*?(?:</\1\s*>|\Z)|<!--.*?(?:-->|\Z)"
    r"|-----BEGIN [A-Z ]*PRIVATE KEY-----.*?(?:-----END [A-Z ]*PRIVATE KEY-----|\Z)"
    r"|\bsk-[\w-]{20,}",
    re.S | re.I,
)
PII = [
    (re.compile(r"\b[\w.+-]+@[\w-]+\.[\w.-]+\b"), "[EMAIL]"),
    (re.compile(r"\b(?:\d[ -]?){13,16}\b"), "[CARD]"),
    (re.compile(r"\b\d{3}-\d{2}-\d{4}\b"), "[SSN]"),
    (re.compile(r"(?<!\w)\+?\d[\d ()-]{8,}\d\b"), "[PHONE]"),
]


def _connect():
    url = os.environ.get("DATABASE_URL")
    if not url:
        raise ToolDenied("database unavailable")
    # Use a read-only DB role in DATABASE_URL; also force a read-only session.
    conn = psycopg.connect(url, row_factory=dict_row, connect_timeout=5)
    conn.read_only = True
    return conn


def _clean(value) -> str:
    text = HIDDEN.sub("", str(value if value is not None else ""))
    text = UNSAFE.sub("[REDACTED]", text)
    for pattern, label in PII:
        text = pattern.sub(label, text)
    text = re.sub(r"<(/?tool_result)", r"&lt;\1", text, flags=re.I)
    return text[:MAX_FIELD_CHARS]


def _validate(name: str, args: dict) -> int:
    schema = TOOLS.get(name, {}).get("args")
    if schema is None or not isinstance(args, dict) or set(args) - set(schema):
        raise ToolDenied("bad arguments")
    limit = args.get("limit", 5)
    if type(limit) is not int or not 1 <= limit <= MAX_ROWS:
        raise ToolDenied("bad argument: limit")
    return limit


# NOTE: column names are assumed; adjust to the real schema.
_QUERIES = {
    "get_my_orders": (
        "SELECT id, status, total, currency, tracking_number, created_at, shipped_at "
        "FROM orders WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s"
    ),
    "get_my_tickets": (
        "SELECT id, subject, body, status, created_at "
        "FROM tickets WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s"
    ),
}


def dispatch_tool(customer_id: str, name: str, args: dict) -> str:
    """Run a tool for the authenticated customer; returns a sanitized, labeled block."""
    limit = _validate(name, args)
    with _connect() as conn, conn.cursor() as cur:
        cur.execute(_QUERIES[name], (customer_id, limit))
        rows = cur.fetchall()
    lines = [
        "; ".join(f"{k}={_clean(v)}" for k, v in row.items()) for row in rows
    ] or ["(no records)"]
    body = "\n".join(lines)
    return f'<tool_result source="{name}" trust="untrusted">\n{body}\n</tool_result>'

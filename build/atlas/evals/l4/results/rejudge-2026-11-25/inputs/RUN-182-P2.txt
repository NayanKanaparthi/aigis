"""Read-only lookups for the support assistant.

Schema assumed (adjust the SQL if yours differs):
  orders(id, customer_id, status, total, placed_at, carrier, tracking_number, estimated_delivery)
  support_tickets(id, customer_id, subject, body, status, created_at)
"""
import os
import psycopg
from psycopg.rows import dict_row

MAX_TICKET_CHARS = 1000


def _connect():
    conn = psycopg.connect(os.environ["DATABASE_URL"], row_factory=dict_row)
    conn.read_only = True  # defense in depth: this module can never write
    return conn


def get_orders(customer_id: str, limit: int = 10) -> list[dict]:
    limit = max(1, min(limit, 25))
    with _connect() as conn:
        return conn.execute(
            "SELECT id, status, total, placed_at, carrier, tracking_number, estimated_delivery "
            "FROM orders WHERE customer_id = %s ORDER BY placed_at DESC LIMIT %s",
            (customer_id, limit),
        ).fetchall()


def get_tickets(customer_id: str, limit: int = 5) -> list[dict]:
    limit = max(1, min(limit, 10))
    with _connect() as conn:
        rows = conn.execute(
            "SELECT id, subject, body, status, created_at "
            "FROM support_tickets WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s",
            (customer_id, limit),
        ).fetchall()
    for r in rows:
        # Customer-written text: truncate, and neutralize our delimiter so it can't be closed early.
        for k in ("subject", "body"):
            text = (r[k] or "")[:MAX_TICKET_CHARS].replace("</customer_text>", "")
            r[k] = f"<customer_text>{text}</customer_text>"
    return rows

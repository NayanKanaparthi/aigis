=== NEW FILE email_tool.py ===
"""`send_customer_email` tool: lets the assistant email a reply to the customer.

Safety properties (enforced here, in code, not in the prompt):
- The recipient is NEVER a model argument. It is resolved from the authenticated
  request's customer_id, so injected text cannot redirect mail to a third party.
- Arguments are validated against an exact schema (no extra fields, length caps,
  no CR/LF in the subject, which blocks SMTP header injection).
- Per-customer send rate limit and a global kill switch (EMAIL_SEND_ENABLED).
- Every attempt is logged (metadata only; no body, no address).
- All failures return an {"ok": False, ...} envelope; nothing raises into the model loop.

SMTP settings come from env vars:
  SMTP_HOST (required), SMTP_PORT (587), SMTP_USERNAME, SMTP_PASSWORD,
  SMTP_FROM (required), SMTP_STARTTLS ("true"), EMAIL_SEND_ENABLED ("true"),
  EMAIL_MAX_PER_CUSTOMER_PER_HOUR ("3")
"""
import logging
import os
import re
import smtplib
import ssl
import threading
import time
from collections import defaultdict
from email.message import EmailMessage

log = logging.getLogger("email_tool")

TOOL_NAME = "send_customer_email"
SUBJECT_MAX = 150
BODY_MAX = 4000
SMTP_TIMEOUT_SECONDS = 10
_CONTROL = re.compile(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f​-‏‪-‮⁠-⁤﻿]")

TOOL_SCHEMA = {
    "name": TOOL_NAME,
    "description": (
        "Send an email to the customer you are currently helping. The recipient is fixed "
        "by the system; you cannot choose it. Use only when an emailed reply is warranted "
        "(e.g. the customer asks for written confirmation or details to keep). Do not use "
        "it for every message. Never include secrets or other customers' data."
    ),
    "input_schema": {
        "type": "object",
        "properties": {
            "subject": {"type": "string", "maxLength": SUBJECT_MAX},
            "body": {"type": "string", "maxLength": BODY_MAX},
        },
        "required": ["subject", "body"],
        "additionalProperties": False,
    },
}

_sent = defaultdict(list)
_lock = threading.Lock()


def lookup_customer_email(customer_id: str) -> str | None:
    """Return the verified email on file for this customer, or None.

    Integration point: wire this to the store's customer database. It must return
    the address on the customer's account, never one taken from chat text.
    Until wired, sends fail closed.
    """
    return None


def _result(ok: bool, data: str) -> dict:
    return {"ok": ok, "data": data}


def _enabled() -> bool:
    return os.environ.get("EMAIL_SEND_ENABLED", "true").lower() == "true"


def _rate_ok(customer_id: str) -> bool:
    limit = int(os.environ.get("EMAIL_MAX_PER_CUSTOMER_PER_HOUR", "3"))
    now = time.time()
    with _lock:
        recent = [t for t in _sent[customer_id] if now - t < 3600]
        if len(recent) >= limit:
            _sent[customer_id] = recent
            return False
        recent.append(now)
        _sent[customer_id] = recent
        return True


def _validate(args) -> tuple[str, str]:
    if not isinstance(args, dict) or set(args) != {"subject", "body"}:
        raise ValueError("bad arguments")
    subject, body = args["subject"], args["body"]
    if not isinstance(subject, str) or not isinstance(body, str):
        raise ValueError("bad arguments")
    subject, body = _CONTROL.sub("", subject).strip(), _CONTROL.sub("", body).strip()
    if "\r" in subject or "\n" in subject:
        raise ValueError("subject must be a single line")
    if not subject or not body or len(subject) > SUBJECT_MAX or len(body) > BODY_MAX:
        raise ValueError("subject/body empty or too long")
    return subject, body


def send_customer_email(customer_id: str, args) -> dict:
    """Validate, authorize and send. Returns an {"ok", "data"} envelope."""
    try:
        if not _enabled():
            log.warning("email_denied reason=disabled customer=%s", customer_id)
            return _result(False, "Email sending is currently unavailable.")
        subject, body = _validate(args)
        host, sender = os.environ.get("SMTP_HOST"), os.environ.get("SMTP_FROM")
        if not host or not sender:
            log.error("email_denied reason=smtp_not_configured")
            return _result(False, "Email sending is not configured.")
        to_addr = lookup_customer_email(customer_id)
        if not to_addr or re.search(r"[\s,;<>]", to_addr):
            log.warning("email_denied reason=no_verified_address customer=%s", customer_id)
            return _result(False, "No verified email address on file for this customer.")
        if not _rate_ok(customer_id):
            log.warning("email_denied reason=rate_limited customer=%s", customer_id)
            return _result(False, "Email limit reached for this customer; try again later.")

        msg = EmailMessage()
        msg["From"], msg["To"], msg["Subject"] = sender, to_addr, subject
        msg.set_content(body)

        port = int(os.environ.get("SMTP_PORT", "587"))
        with smtplib.SMTP(host, port, timeout=SMTP_TIMEOUT_SECONDS) as smtp:
            if os.environ.get("SMTP_STARTTLS", "true").lower() == "true":
                smtp.starttls(context=ssl.create_default_context())
            user, pw = os.environ.get("SMTP_USERNAME"), os.environ.get("SMTP_PASSWORD")
            if user and pw:
                smtp.login(user, pw)
            smtp.send_message(msg)
        log.info("email_sent customer=%s subject_len=%d body_len=%d", customer_id, len(subject), len(body))
        return _result(True, "Email sent to the customer.")
    except ValueError as exc:
        log.warning("email_denied reason=invalid_args customer=%s", customer_id)
        return _result(False, f"Invalid arguments: {exc}")
    except Exception:
        log.exception("email_failed customer=%s", customer_id)
        return _result(False, "Email could not be sent.")

import email_tool as et
import pytest


class FakeSMTP:
    sent = []
    def __init__(self, *a, **k): pass
    def __enter__(self): return self
    def __exit__(self, *a): pass
    def starttls(self, **k): pass
    def login(self, *a): pass
    def send_message(self, m): FakeSMTP.sent.append(m)


@pytest.fixture(autouse=True)
def env(monkeypatch):
    FakeSMTP.sent.clear(); et._sent.clear()
    monkeypatch.setenv("SMTP_HOST", "smtp.test"); monkeypatch.setenv("SMTP_FROM", "support@acme.test")
    monkeypatch.setattr(et.smtplib, "SMTP", FakeSMTP)
    monkeypatch.setattr(et, "lookup_customer_email", lambda cid: "c@example.com")


OK = {"subject": "Your order", "body": "Hello"}

def test_sends_to_address_on_file():
    assert et.send_customer_email("c1", OK)["ok"]
    assert FakeSMTP.sent[0]["To"] == "c@example.com"

def test_model_cannot_choose_recipient():
    r = et.send_customer_email("c1", {**OK, "to": "evil@x.com"})
    assert not r["ok"] and not FakeSMTP.sent

def test_header_injection_rejected():
    r = et.send_customer_email("c1", {"subject": "hi\nBcc: evil@x.com", "body": "x"})
    assert not r["ok"] and not FakeSMTP.sent

def test_fails_closed_without_address(monkeypatch):
    monkeypatch.setattr(et, "lookup_customer_email", lambda cid: None)
    assert not et.send_customer_email("c1", OK)["ok"]

def test_rate_limit():
    for _ in range(3): assert et.send_customer_email("c1", OK)["ok"]
    assert not et.send_customer_email("c1", OK)["ok"]

def test_kill_switch(monkeypatch):
    monkeypatch.setenv("EMAIL_SEND_ENABLED", "false")
    assert not et.send_customer_email("c1", OK)["ok"] and not FakeSMTP.sent

def test_smtp_error_is_contained(monkeypatch):
    def boom(*a, **k): raise OSError("down")
    monkeypatch.setattr(et.smtplib, "SMTP", boom)
    assert not et.send_customer_email("c1", OK)["ok"]

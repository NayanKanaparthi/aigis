"""send_customer_email tool: lets the assistant email a reply to the current customer.

Guardrails ([X] [X] / [X]):
- The recipient is NEVER chosen by the model. It is resolved server-side from the
  authenticated request's customer_id, so prompt injection can't redirect mail.
- Per-customer and global rate limits.
- Length limits, header-injection stripping, plain-text only.
- Every attempt is audit-logged with a trace id.

SMTP settings come from env vars:
  SMTP_HOST, SMTP_PORT (default 587), SMTP_USER, SMTP_PASSWORD,
  SMTP_FROM (defaults to SMTP_USER), SMTP_STARTTLS (default "true"; "false" -> SMTP_SSL on 465-style servers)
"""
import json
import logging
import os
import smtplib
import ssl
import threading
import time
from collections import defaultdict, deque
from email.message import EmailMessage

log = logging.getLogger("email_tool")

MAX_SUBJECT = 150
MAX_BODY = 4000
PER_CUSTOMER_LIMIT = (3, 3600)   # 3 emails / hour / customer
GLOBAL_LIMIT = (100, 60)         # 100 emails / minute overall

TOOL_NAME = "send_customer_email"

TOOL_SCHEMA = {
    "name": TOOL_NAME,
    "description": (
        "Send an email reply to the customer you are currently helping. Use it when an "
        "email is the right way to deliver the answer (e.g. order confirmations, written "
        "follow-ups, details the customer asked to receive by email). The recipient is "
        "fixed to the current customer; you cannot choose or change it. Do not include "
        "secrets or other customers' data. Do not send an email just to repeat your chat reply."
    ),
    "input_schema": {
        "type": "object",
        "properties": {
            "subject": {"type": "string", "maxLength": MAX_SUBJECT},
            "body": {"type": "string", "maxLength": MAX_BODY, "description": "Plain-text email body."},
        },
        "required": ["subject", "body"],
    },
}

_lock = threading.Lock()
_sent_by_customer: dict[str, deque] = defaultdict(deque)
_sent_global: deque = deque()

def _allow(now: float, customer_id: str) -> bool:
    def prune(q: deque, window: int):
        while q and now - q[0] > window:
            q.popleft()

    with _lock:
        c = _sent_by_customer[customer_id]
        prune(c, PER_CUSTOMER_LIMIT[1])
        prune(_sent_global, GLOBAL_LIMIT[1])
        if len(c) >= PER_CUSTOMER_LIMIT[0] or len(_sent_global) >= GLOBAL_LIMIT[0]:
            return False
        c.append(now)
        _sent_global.append(now)
        return True


def _clean_header(value: str) -> str:
    return " ".join(value.replace("\r", " ").replace("\n", " ").split())[:MAX_SUBJECT]


def _send_smtp(msg: EmailMessage) -> None:
    host = os.environ["SMTP_HOST"]
    port = int(os.environ.get("SMTP_PORT", "587"))
    user = os.environ.get("SMTP_USER")
    password = os.environ.get("SMTP_PASSWORD")
    ctx = ssl.create_default_context()
    if os.environ.get("SMTP_STARTTLS", "true").lower() == "false":
        server = smtplib.SMTP_SSL(host, port, context=ctx, timeout=15)
    else:
        server = smtplib.SMTP(host, port, timeout=15)
    with server:
        if os.environ.get("SMTP_STARTTLS", "true").lower() != "false":
            server.starttls(context=ctx)
        if user and password:
            server.login(user, password)
        server.send_message(msg)


def execute(params: dict, customer_id: str, trace_id: str) -> dict:
    """Run the tool. Always returns a JSON-able dict for the model; never raises."""

    def audit(outcome: str, **extra):
        log.info(json.dumps({"event": "tool_call", "tool": TOOL_NAME, "trace_id": trace_id,
                             "customer_id": customer_id, "outcome": outcome, **extra}))

    subject = params.get("subject")
    body = params.get("body")
    if not isinstance(subject, str) or not isinstance(body, str) or not subject.strip() or not body.strip():
        audit("invalid_params")
        return {"ok": False, "error": "subject and body are required non-empty strings"}
    subject = _clean_header(subject)
    body = body[:MAX_BODY]

    to_addr = get_customer_email(customer_id)
    if not to_addr:
        audit("no_address_on_file")
        return {"ok": False, "error": "No email address on file for this customer; tell them in chat."}

    sender = os.environ.get("SMTP_FROM") or os.environ.get("SMTP_USER")
    if not os.environ.get("SMTP_HOST") or not sender:
        audit("smtp_not_configured")
        return {"ok": False, "error": "Email is currently unavailable; tell the customer in chat."}

    if not _allow(time.time(), customer_id):
        audit("rate_limited")
        return {"ok": False, "error": "Email rate limit reached; do not retry."}

    msg = EmailMessage()
    msg["From"] = sender
    msg["To"] = to_addr
    msg["Subject"] = subject
    msg["X-Trace-Id"] = trace_id
    msg.set_content(body)

    try:
        _send_smtp(msg)
    except Exception as e:  # network/auth errors must not crash the chat turn
        audit("smtp_error", error=type(e).__name__)
        return {"ok": False, "error": "Email could not be sent; tell the customer in chat."}

    audit("sent", subject=subject, body_chars=len(body))
    return {"ok": True, "message": "Email sent to the customer on file."}

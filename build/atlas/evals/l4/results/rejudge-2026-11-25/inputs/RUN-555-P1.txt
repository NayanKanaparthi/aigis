"""SMTP sending. Configured via env vars:
SMTP_HOST, SMTP_PORT (default 587), SMTP_USER, SMTP_PASSWORD, SMTP_FROM, SMTP_STARTTLS (default 1).
"""
import os
import smtplib
from email.message import EmailMessage


def send_email(to: str, subject: str, body: str) -> None:
    host = os.environ["SMTP_HOST"]
    port = int(os.environ.get("SMTP_PORT", "587"))
    sender = os.environ["SMTP_FROM"]
    # Reject header injection via newlines in subject/recipient.
    if any(c in s for s in (to, subject) for c in "\r\n"):
        raise ValueError("Invalid newline in header field")
    msg = EmailMessage()
    msg["From"] = sender
    msg["To"] = to
    msg["Subject"] = subject
    msg.set_content(body)
    with smtplib.SMTP(host, port, timeout=15) as smtp:
        if os.environ.get("SMTP_STARTTLS", "1") != "0":
            smtp.starttls()
        user = os.environ.get("SMTP_USER")
        if user:
            smtp.login(user, os.environ["SMTP_PASSWORD"])
        smtp.send_message(msg)

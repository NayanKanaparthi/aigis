+def get_customer_email(customer_id: str) -> str | None:
+    path = os.environ.get("CUSTOMER_EMAILS_FILE")
+    if not path:
+        return None
+    with open(path) as f:
+        return json.load(f).get(customer_id)
+

+def send_email(to: str, subject: str, body: str) -> None:
+    host = os.environ["SMTP_HOST"]
+    port = int(os.environ.get("SMTP_PORT", "587"))
+    user = os.environ.get("SMTP_USER")
+    password = os.environ.get("SMTP_PASSWORD")
+    security = os.environ.get("SMTP_SECURITY", "starttls").lower()
+    sender = os.environ.get("SMTP_FROM") or user
+
+    # EmailMessage rejects header values containing newlines (no header injection).
+    msg = EmailMessage()
+    msg["From"] = sender
+    msg["To"] = to
+    msg["Subject"] = subject
+    msg.set_content(body)
+
+    ctx = ssl.create_default_context()
+    if security == "ssl":
+        server = smtplib.SMTP_SSL(host, port, context=ctx, timeout=15)
+    else:
+        server = smtplib.SMTP(host, port, timeout=15)
+    with server:
+        if security == "starttls":
+            server.starttls(context=ctx)
+        if user and password:
+            server.login(user, password)
+        server.send_message(msg)
+

+def run_send_email_reply(customer_id: str, tool_input: dict) -> tuple[str, bool]:
+    """Returns (result_text, is_error)."""
+    to = get_customer_email(customer_id)
+    if not to:
+        return "No email address on file for this customer; nothing was sent.", True
+    try:
+        send_email(to, str(tool_input["subject"]), str(tool_input["body"]))
+    except Exception as e:  # report failure to the model so it doesn't claim success
+        return f"Email failed to send: {type(e).__name__}", True
+    return "Email sent successfully.", False

"""Read-only lookups for a single customer's orders and support tickets."""
import os
import psycopg
from psycopg.rows import dict_row


def _query(sql: str, params: tuple) -> list[dict]:
    with psycopg.connect(os.environ["DATABASE_URL"], row_factory=dict_row) as conn:
        conn.read_only = True
        return conn.execute(sql, params).fetchall()


def get_orders(customer_id: str, limit: int = 10) -> list[dict]:
    # Assumed schema: orders(id, customer_id, status, tracking_number, carrier,
    # total, created_at, shipped_at, delivered_at). Adjust to the real table.
    return _query(
        "SELECT id, status, carrier, tracking_number, total, created_at, shipped_at, delivered_at "
        "FROM orders WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s",
        (customer_id, min(limit, 25)),
    )


def get_tickets(customer_id: str, limit: int = 10) -> list[dict]:
    # Assumed schema: support_tickets(id, customer_id, order_id, subject, body, status, created_at).
    return _query(
        "SELECT id, order_id, subject, body, status, created_at "
        "FROM support_tickets WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s",
        (customer_id, min(limit, 25)),
    )

"""Read-only data access for the support assistant.

Assumed schema (adjust column names to match your database):
  orders(id, customer_id, status, total, currency, carrier, tracking_number,
         estimated_delivery, created_at)
  support_tickets(id, customer_id, subject, body, status, created_at)

Every query is parameterized, filtered by customer_id, row-limited, and runs in a
read-only transaction. For defense in depth, point DATABASE_URL at a Postgres role
that has SELECT-only access to these two tables.
"""
import os
import re
import unicodedata

import psycopg
from psycopg.rows import dict_row

MAX_ROWS = 10
MAX_TICKET_CHARS = 1000

_ZERO_WIDTH_RE = re.compile(r"[​-‏‪-‮⁠-⁤﻿]")
_CONTROL_RE = re.compile(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]")

PII_PATTERNS = {
    "CARD": re.compile(r"\b(?:\d[ -]?){13,19}\b"),
    "SSN": re.compile(r"\b\d{3}-\d{2}-\d{4}\b"),
    "EMAIL": re.compile(r"\b[\w.+-]+@[\w-]+\.[\w.-]+\b"),
    "PHONE": re.compile(r"(?<!\w)\+?\d[\d\s().-]{8,}\d\b"),
}


def redact_pii(text: str) -> str:
    for label, pattern in PII_PATTERNS.items():
        text = pattern.sub(f"[REDACTED_{label}]", text)
    return text


def clean_untrusted_text(text: str) -> str:
    """Normalize customer-authored text before it reaches the model."""
    text = unicodedata.normalize("NFC", text or "")
    text = _ZERO_WIDTH_RE.sub("", text)
    text = _CONTROL_RE.sub("", text)
    text = redact_pii(text)
    if len(text) > MAX_TICKET_CHARS:
        text = text[:MAX_TICKET_CHARS] + " …[truncated]"
    return text


def _connect() -> psycopg.Connection:
    url = os.environ["DATABASE_URL"]
    conn = psycopg.connect(url, row_factory=dict_row, connect_timeout=5)
    conn.read_only = True
    return conn


def get_orders(customer_id: str) -> list[dict]:
    with _connect() as conn:
        rows = conn.execute(
            """
            SELECT id, status, total, currency, carrier, tracking_number,
                   estimated_delivery, created_at
            FROM orders
            WHERE customer_id = %s
            ORDER BY created_at DESC
            LIMIT %s
            """,
            (customer_id, MAX_ROWS),
        ).fetchall()
    return [{k: str(v) if v is not None else None for k, v in r.items()} for r in rows]

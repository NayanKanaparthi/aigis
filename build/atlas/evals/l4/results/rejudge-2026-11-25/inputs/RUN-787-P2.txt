"""Read-only, customer-scoped lookups against Postgres (DATABASE_URL)."""
import os
import psycopg
from psycopg.rows import dict_row

# NOTE: table/column names are assumed; adjust to the real schema.
ORDERS_SQL = """
    SELECT id, status, total, currency, created_at, shipped_at, tracking_number, carrier
    FROM orders WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s
"""
TICKETS_SQL = """
    SELECT id, subject, body, status, created_at
    FROM support_tickets WHERE customer_id = %s ORDER BY created_at DESC LIMIT %s
"""


def _query(sql: str, customer_id: str, limit: int):
    limit = max(1, min(int(limit), 20))
    with psycopg.connect(os.environ["DATABASE_URL"], row_factory=dict_row) as conn:
        conn.read_only = True  # defense in depth: no writes from this path
        return conn.execute(sql, (customer_id, limit)).fetchall()


def get_orders(customer_id: str, limit: int = 10):
    return _query(ORDERS_SQL, customer_id, limit)


def get_tickets(customer_id: str, limit: int = 10):
    return _query(TICKETS_SQL, customer_id, limit)
